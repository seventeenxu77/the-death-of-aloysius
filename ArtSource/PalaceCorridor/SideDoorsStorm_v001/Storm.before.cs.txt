using System.Collections.Generic;
using UnityEngine;

namespace Aloysius.Level1
{
    [DefaultExecutionOrder(650)]
    public sealed class Level1PalaceStorm : MonoBehaviour
    {
        public enum Layer { Sky, Distant, Garden, Architecture, Cover }
        [System.Serializable] public class LightLayer
        {
            public Layer layer;
            public Renderer[] renderers;
            [Range(0,1)] public float darkExposure=0f;
            [Range(0,2)] public float flashExposure=.85f;
            [Range(0,1)] public float blueFill=.12f;
            [Range(0,1)] public float rim=.2f;
        }
        [Header("镜厅引用")]
        [SerializeField] Level1PalaceCorridor corridor;
        [SerializeField] Camera view;
        [SerializeField] LightLayer[] layers;
        [SerializeField] Renderer[] windowProjections;
        [SerializeField] Renderer rain;
        [SerializeField] LineRenderer lightningBolt;
        [SerializeField] Transform leftBoundary,rightBoundary,ground;
        [Header("启用与黑暗等待（秒）")]
        [SerializeField] bool enableStorm=true;
        [SerializeField,Min(0)] float firstFlashDelay=3f;
        [SerializeField,Min(.5f),Tooltip("一次闪电余光结束后，到下一次闪电开始前的固定黑暗时长（秒）。")] float darkDuration=3f;
        [SerializeField,Min(.01f)] float darknessFade=.7f;
        [SerializeField,Tooltip("0 每次游戏随机；其他数值可固定随机序列。")] int randomSeed;
        [Header("一次闪电的节奏（秒）")]
        [SerializeField,Min(.02f)] float weakFlashDuration=.06f;
        [SerializeField,Range(0,1)] float weakFlashStrength=.3f;
        [SerializeField,Min(.02f)] float interFlashPause=.12f;
        [SerializeField,Min(.03f)] float strongFlashDuration=.15f;
        [SerializeField,Min(.05f)] float afterglowDuration=.5f;
        [SerializeField,Min(0)] float indoorLightDelay=.025f;
        [SerializeField] Vector2 intensityRange=new Vector2(.72f,1f);
        [SerializeField] Vector2 durationVariation=new Vector2(.85f,1.15f);
        [Header("冷光与人物剪影")]
        [SerializeField] Color lightningTint=new Color(.60f,.74f,1f,1);
        [SerializeField,Range(0,1)] float darkCharacterExposure=0f;
        [SerializeField,Range(0,1)] float flashCharacterExposure=.58f;
        [SerializeField,Range(0,.3f)] float idleCharacterRim=0f;
        [SerializeField,Range(0,1)] float flashCharacterRim=.45f;
        [Header("出管口外壳亮度")]
        [SerializeField,Range(0,1)] float darkPipeExposure=0f;
        [SerializeField,Range(0,1)] float flashPipeExposure=.58f;
        [Header("窗格投影")]
        [SerializeField,Range(0,1)] float projectionStrength=.4f;
        [Header("人物附近微光（仅镜厅）")]
        [SerializeField] bool enablePlayerGlow=true;
        [SerializeField,Min(.05f),Tooltip("世界单位，光照到此半径完全消失。")] float playerGlowRadius=1.1f;
        [SerializeField] Vector2 playerGlowOffset=new Vector2(0,.65f);
        [SerializeField,Range(.05f,.95f),Tooltip("开始柔和衰减的位置，占半径的比例。")] float playerGlowSoftness=.2f;
        [SerializeField] Color playerGlowTint=new Color(.60f,.70f,.85f,1);
        [SerializeField,Range(0,.5f)] float playerGlowEnvironment=.085f;
        [SerializeField,Range(0,.5f)] float playerGlowCharacter=.12f;
        [SerializeField,Range(0,.3f)] float playerGlowRim=.055f;
        [Header("窗外细雨与闪电枝条")]
        [SerializeField] bool showRain=true;
        [SerializeField,Range(0,1)] float rainOpacity=0f;
        [SerializeField,Range(0,1)] float rainFlashBoost=.2f;
        [SerializeField,Min(.1f)] float rainSpeed=3.2f;
        [SerializeField,Range(-1,1)] float rainSlant=.16f;
        [SerializeField] bool showLightningBolt=true;
        [SerializeField,Range(0,1)] float boltStrength=.7f;
        [SerializeField,Min(.001f)] float boltWidth=.023f;
        readonly Dictionary<Renderer,LightLayer> lookup=new Dictionary<Renderer,LightLayer>();
        System.Random random;
        MaterialPropertyBlock block;
        float weight,wait,clock,amplitude=1,durationScale=1;
        bool wasActive,flashing;
        public float OutsideFlash { get; private set; }
        public float InsideFlash { get; private set; }
        public float Weight=>weight;
        public Color Tint=>lightningTint;
        public float CharacterRim=>weight*(idleCharacterRim+InsideFlash*flashCharacterRim);
        public float CharacterExposure(float original)=>Mathf.Lerp(original,Mathf.Lerp(darkCharacterExposure,flashCharacterExposure,InsideFlash),weight);
        public float PipeExposure(float original)=>Mathf.Lerp(original,Mathf.Lerp(darkPipeExposure,flashPipeExposure,InsideFlash),weight);

        public void ApplyPlayerGlow(MaterialPropertyBlock properties,bool character,float visibility=1f)
        {
            var anchor=corridor?corridor.PlayerLightAnchor:null;
            float amount=enablePlayerGlow && anchor ? weight*Mathf.Clamp01(visibility):0;
            Vector2 center=anchor?(Vector2)anchor.position+playerGlowOffset:Vector2.zero;
            properties.SetVector("_PlayerGlowArea",new Vector4(center.x,center.y,Mathf.Max(.05f,playerGlowRadius),Mathf.Clamp(playerGlowSoftness,.05f,.95f)));
            properties.SetColor("_PlayerGlowTint",playerGlowTint);
            properties.SetVector("_PlayerGlowStrength",new Vector4(amount*(character?playerGlowCharacter:playerGlowEnvironment),character?amount*playerGlowRim:0,0,0));
        }

        void Awake()
        {
            random=new System.Random(randomSeed==0?System.Environment.TickCount:randomSeed);
            RebuildLayers();wait=firstFlashDelay;
        }
        void RebuildLayers()
        {
            lookup.Clear();if(layers==null)return;
            foreach(var layer in layers)if(layer!=null && layer.renderers!=null)
                foreach(var r in layer.renderers)if(r)lookup[r]=layer;
        }
        void OnValidate(){if(Application.isPlaying)RebuildLayers();}
        float Pick(Vector2 range,float minimum)=>Mathf.Lerp(Mathf.Max(minimum,Mathf.Min(range.x,range.y)),Mathf.Max(minimum,Mathf.Max(range.x,range.y)),(float)random.NextDouble());
        void Update()
        {
            bool active=enableStorm && corridor && corridor.Arrived;
            float dt=Mathf.Max(0,Time.deltaTime);
            weight=Mathf.MoveTowards(weight,active?1:0,dt/Mathf.Max(.01f,darknessFade));
            if(!active)
            {
                wasActive=flashing=false;OutsideFlash=InsideFlash=0;clock=0;wait=firstFlashDelay;return;
            }
            if(!wasActive){wasActive=true;wait=Mathf.Max(0,firstFlashDelay);}
            if(!flashing)
            {
                wait-=dt;OutsideFlash=InsideFlash=0;
                if(wait>0)return;
                flashing=true;clock=0;amplitude=Pick(intensityRange,0);durationScale=Pick(durationVariation,.1f);PlaceBolt();
            }
            clock+=dt;
            OutsideFlash=Envelope(clock/durationScale)*amplitude;
            InsideFlash=Envelope((clock-indoorLightDelay)/durationScale)*amplitude;
            float total=weakFlashDuration+interFlashPause+strongFlashDuration+afterglowDuration;
            if(clock>=total*durationScale+indoorLightDelay)
            {flashing=false;OutsideFlash=InsideFlash=0;wait=Mathf.Max(.5f,darkDuration);}
        }
        float Envelope(float t)
        {
            if(t<0)return 0;
            float weak=Mathf.Max(.02f,weakFlashDuration);
            if(t<weak)return weakFlashStrength*Mathf.Sin(Mathf.PI*t/weak);
            t-=weak+Mathf.Max(.02f,interFlashPause);
            if(t<0)return 0;
            float strong=Mathf.Max(.03f,strongFlashDuration);
            if(t<strong)
            {
                float p=t/strong;
                return p<.12f?Mathf.SmoothStep(0,1,p/.12f):Mathf.Lerp(1,.6f,(p-.12f)/.88f);
            }
            t-=strong;
            return .6f*(1-Mathf.SmoothStep(0,1,Mathf.Clamp01(t/Mathf.Max(.05f,afterglowDuration))));
        }
        public void ApplyLayer(Renderer visual,MaterialPropertyBlock properties,float originalExposure)
        {
            float exposure=Mathf.Lerp(originalExposure,originalExposure*InsideFlash,weight),lift=0,rim=0;
            if(lookup.TryGetValue(visual,out var layer))
            {
                float flash=(layer.layer==Layer.Sky || layer.layer==Layer.Distant || layer.layer==Layer.Garden)?OutsideFlash:InsideFlash;
                exposure=Mathf.Lerp(originalExposure,Mathf.Lerp(layer.darkExposure,layer.flashExposure,flash),weight);
                lift=weight*flash*layer.blueFill;rim=weight*flash*layer.rim;
            }
            properties.SetFloat("_Exposure",exposure);
            properties.SetFloat("_ContactLightVisibility",Mathf.Lerp(1,InsideFlash,weight));
            properties.SetColor("_LightningTint",lightningTint);
            properties.SetFloat("_LightningLift",lift);properties.SetFloat("_LightningRim",rim);
            // A local indoor light must not illuminate distant scenery through the windows.
            bool nearby=layer!=null && (layer.layer==Layer.Architecture || layer.layer==Layer.Cover);
            ApplyPlayerGlow(properties,false,nearby?1:0);
        }
        void LateUpdate()
        {
            if(block==null)block=new MaterialPropertyBlock();
            if(windowProjections!=null)foreach(var r in windowProjections)SetOverlay(r,weight*InsideFlash*projectionStrength,0);
            SetOverlay(rain,showRain?weight*(rainOpacity+OutsideFlash*rainFlashBoost):0,1);
            if(lightningBolt)lightningBolt.widthMultiplier=boltWidth;
            SetOverlay(lightningBolt,showLightningBolt?weight*OutsideFlash*boltStrength:0,2);
        }
        void SetOverlay(Renderer r,float strength,float mode)
        {
            if(!r)return;r.enabled=strength>.0001f;
            r.GetPropertyBlock(block);block.SetFloat("_Strength",strength);block.SetFloat("_Mode",mode);
            block.SetColor("_LightTint",lightningTint);block.SetVector("_Rain",new Vector4(rainSpeed,rainSlant,Time.time,0));
            block.SetVector("_WorldClip",new Vector4(leftBoundary.position.x,rightBoundary.position.x,-1000,1000));r.SetPropertyBlock(block);
        }
        void PlaceBolt()
        {
            if(!lightningBolt || !view || !ground)return;
            float spread=view.orthographicSize*view.aspect*.65f;
            float x=Mathf.Clamp(view.transform.position.x+Pick(new Vector2(-spread,spread),-1000),leftBoundary.position.x+.5f,rightBoundary.position.x-.5f);
            const int count=9;lightningBolt.positionCount=count;lightningBolt.useWorldSpace=false;
            float bend=0;
            for(int i=0;i<count;i++)
            {
                bend+=Pick(new Vector2(-.22f,.22f),-1);
                var world=new Vector3(x+bend,ground.position.y+Mathf.Lerp(6.2f,1.4f,i/(float)(count-1)),4);
                lightningBolt.SetPosition(i,lightningBolt.transform.InverseTransformPoint(world));
            }
        }
        void OnDisable()
        {
            weight=OutsideFlash=InsideFlash=0;wasActive=flashing=false;
            if(windowProjections!=null)foreach(var r in windowProjections)if(r)r.enabled=false;
            if(rain)rain.enabled=false;if(lightningBolt)lightningBolt.enabled=false;
        }
    }
}
